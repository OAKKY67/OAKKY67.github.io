03 // GRAPHICS PROGRAMMING

GLSL SHADER

ตัวอย่างการใช้ Fragment Shader เพื่อสร้าง procedural cyberpunk material แบบ real-time 

uniform float uTime;
uniform vec2 uMouse;

varying vec2 vUv;

void main() {

    vec2 uv = vUv;

    float wave =
        sin(uv.y * 20.0 + uTime * 2.0);

    float pulse =
        sin(uTime * 3.0) * 0.5 + 0.5;

    vec3 cyan =
        vec3(0.0, 1.0, 1.0);

    vec3 pink =
        vec3(1.0, 0.0, 0.4);

    vec3 color =
        mix(cyan, pink, uv.x);

    color *=
        0.5 + wave * 0.3;

    color +=
        pulse * 0.1;

    gl_FragColor =
        vec4(color, 1.0);
}
