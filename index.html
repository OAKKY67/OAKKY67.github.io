document.querySelector('a-scene').addEventListener('arReady', () => { const THREE = AFRAME.THREE; const texture = new THREE.TextureLoader().load( 'https://oakky67.github.io/face.png', (texture) => { texture.colorSpace = THREE.SRGBColorSpace; const mesh = document.querySelector('#face') .getObject3D('mesh'); if (!mesh) { console.error('ไม่พบ Face Mesh'); return; } mesh.material.dispose(); mesh.material = new THREE.MeshBasicMaterial({ map: texture, transparent: true, opacity: 1, side: THREE.DoubleSide, depthWrite: false }); mesh.frustumCulled = false; }, undefined, (error) => { console.error('โหลด face.png ไม่สำเร็จ', error); } ); });
